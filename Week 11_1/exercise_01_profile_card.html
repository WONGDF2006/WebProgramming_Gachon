E1 · Build a ProfileCard component

Task: a ProfileCard that takes name, role, email, and an optional avatarUrl with a default. The code below runs live on this page.

// Default avatar: an inline SVG circle, so the card works offline too
const DEFAULT_AVATAR = 'data:image/svg+xml,' + encodeURIComponent(
  '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 80 80"><circle cx="40" cy="40" r="40" fill="#0e7490"/></svg>'
);

function ProfileCard({ name, role, email, avatarUrl = DEFAULT_AVATAR }) {
  return (
    <div style={{ border: '1px solid #cbd5e1', borderRadius: 8, padding: 16, maxWidth: 300, background: '#fff' }}>
      <img src={avatarUrl} alt="" width="80" height="80" style={{ borderRadius: '50%' }} />
      <h2 style={{ margin: '8px 0 0' }}>{name}</h2>
      <p style={{ margin: '0 0 8px' }}>{role}</p>
      <a href={`mailto:${email}`}>{email}</a>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(
  <ProfileCard name="Dr. Toshpulatov" role="Professor" email="muhiddin@gachon.ac.kr" />
);
