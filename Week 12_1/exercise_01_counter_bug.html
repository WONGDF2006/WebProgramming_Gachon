Demo 01 · CounterBug (a regular variable)

The broken counter from the lecture. The click handler changes a plain let variable. The log below shows the variable going up, but the count on screen never changes.

const log = document.getElementById('log');

function CounterBug() {
  let count = 0;   // regular variable: React can't see it

  function handleClick() {
    count++;
    log.textContent += ' ' + count;   // stands in for console.log
  }

  return (
    <div>
      <p>Count: {count}</p>
      <button type="button" onClick={handleClick}>+1</button>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<CounterBug />);
