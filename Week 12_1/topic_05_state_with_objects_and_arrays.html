State with objects & arrays

Never mutate. Always create a new object or array. React compares the old and new value by reference (Object.is). If you change the same object and pass it back, React sees “no change” and skips the re-render.

Objects: the spread operator

function ProfileForm() {
  const [user, setUser] = useState({ name: '', email: '', bio: '' });

  // ✗ WRONG: mutation, same object reference
  // user.name = 'Kim Minho';
  // setUser(user);

  // ✓ RIGHT: copy everything, then overwrite one field
  const handleChange = (field, value) => {
    setUser(prev => ({ ...prev, [field]: value }));
  };

  return (
    <form>
      <input value={user.name}  onChange={e => handleChange('name',  e.target.value)} />
      <input value={user.email} onChange={e => handleChange('email', e.target.value)} />
      <textarea value={user.bio} onChange={e => handleChange('bio',  e.target.value)} />
    </form>
  );
}

function TodoList() {
  const [todos, setTodos] = useState([]);
  const [input, setInput] = useState('');

  const addTodo = () => {
    if (!input.trim()) return;
    setTodos(prev => [...prev, { id: Date.now(), text: input, done: false }]);
    setInput('');
  };
  const removeTodo = (id) => setTodos(prev => prev.filter(t => t.id !== id));
  const toggleTodo = (id) =>
    setTodos(prev => prev.map(t => t.id === id ? { ...t, done: !t.done } : t));
  // ...
}

Key points

  • Objects: { ...prev, field: value }.
  • Arrays: spread to add, filter to remove, map to change one item.
  • push returns a number and mutates in place: never pass it to a setter.
