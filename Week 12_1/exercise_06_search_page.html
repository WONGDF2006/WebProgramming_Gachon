Demo 06 · SearchPage (lifting state up)

SearchPage owns query and results. SearchBar sends each change up through onSearch; ResultList only reads props.

const FRUITS = ['Apple', 'Apricot', 'Banana', 'Blueberry', 'Cherry', 'Grape', 'Kiwi', 'Lemon', 'Mango', 'Orange', 'Peach', 'Pear', 'Strawberry', 'Watermelon'];

// Stands in for a server call
const fetchResults = (q) =>
  Promise.resolve(FRUITS.filter((f) => f.toLowerCase().includes(q.trim().toLowerCase())));

function SearchPage() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState(FRUITS);

  const handleSearch = async (q) => {
    setQuery(q);
    const data = await fetchResults(q);
    setResults(data);
  };

  return (
    <>
      <SearchBar onSearch={handleSearch} />
      <ResultList query={query} results={results} />
    </>
  );
}

function SearchBar({ onSearch }) {
  const [input, setInput] = useState('');
  return (
    <div>
      <label htmlFor="fruit-search">Search fruit</label>
      <input id="fruit-search" value={input} onChange={(e) => {
        setInput(e.target.value);
        onSearch(e.target.value);
      }} />
    </div>
  );
}

function ResultList({ query, results }) {
  return (
    <div>
      <p aria-live="polite">{results.length} result{results.length === 1 ? '' : 's'}{query && <> for “{query}”</>}</p>
      <ul>{results.map((r) => <li key={r}>{r}</li>)}</ul>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<SearchPage />);

Live result

fetchResults here filters a local list instead of calling a server, so the demo works offline.
