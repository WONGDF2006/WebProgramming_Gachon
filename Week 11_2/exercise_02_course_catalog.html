E2 · Course Catalog

A CourseCard rendered from an array, with a badge that changes by the required prop.

function CourseCard({ title, instructor, credits, required }) {
  return (
    <article style={{ padding: 16, border: '1px solid #cbd5e1', borderRadius: 12, background: '#fff' }}>
      <h2 style={{ margin: '0 0 6px', fontSize: '1.05rem' }}>{title}</h2>
      <p style={{ margin: '0 0 4px' }}>Instructor: {instructor}</p>
      <p style={{ margin: '0 0 10px' }}>Credits: {credits}</p>
      <span style={{
        padding: '3px 10px', borderRadius: 999, fontSize: '.8rem', fontWeight: 700,
        color: required ? '#991b1b' : '#334155', background: required ? '#fee2e2' : '#e2e8f0',
      }}>
        {required ? '필수 · Required' : '선택 · Elective'}
      </span>
    </article>
  );
}

const courses = [
  { id: 1, title: 'Web Programming', instructor: 'Dr. Toshpulatov', credits: 3, required: true },
  { id: 2, title: 'Database Systems', instructor: 'Dr. Kim', credits: 3, required: true },
  { id: 3, title: 'Mobile App Development', instructor: 'Dr. Lee', credits: 2, required: false },
];

function CourseCatalog() {
  return (
    <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(200px, 1fr))', gap: 14 }}>
      {courses.map((c) => (
        <CourseCard key={c.id} title={c.title} instructor={c.instructor} credits={c.credits} required={c.required} />
      ))}
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<CourseCatalog />);
