E9 · Lab Profile Form

One state object for several inputs, updated with the input's name, plus validation on submit.

function LabProfile() {
  const [profile, setProfile] = useState({ name: '', email: '' });
  const [message, setMessage] = useState('');

  function handleChange(e) {
    const { name, value } = e.target;
    setProfile({ ...profile, [name]: value });   // update just this field
  }

  function handleSubmit(e) {
    e.preventDefault();
    if (!profile.name.trim() || !profile.email.trim()) setMessage('Please fill in both fields.');
    else if (!/^[^\s@]+@[^\s@]+$/.test(profile.email)) setMessage('Please enter a valid email address.');
    else setMessage(`Thanks, ${profile.name.trim()}: profile submitted.`);
  }

  return (
    <form onSubmit={handleSubmit} noValidate>
      <p><label htmlFor="lab-name">Name </label>
        <input id="lab-name" name="name" type="text" value={profile.name} onChange={handleChange} /></p>
      <p><label htmlFor="lab-email">Email </label>
        <input id="lab-email" name="email" type="email" value={profile.email} onChange={handleChange} /></p>
      <button type="submit">Submit profile</button>
      <p role="status">{message}</p>
    </form>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<LabProfile />);

[name]: value is a computed key: the same handler updates name or email depending on which input changed.
