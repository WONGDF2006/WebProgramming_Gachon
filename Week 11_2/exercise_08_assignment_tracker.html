E8 · Assignment Tracker

Toggle tasks done, filter by status, and clear the finished ones. The filtered list is derived from state, not stored.

function AssignmentTracker() {
  const [tasks, setTasks] = useState([
    { id: 1, title: 'Finish Task 11-A', done: false },
    { id: 2, title: 'Practise useState', done: true },
    { id: 3, title: 'Deploy the portfolio', done: false },
  ]);
  const [filter, setFilter] = useState('all');

  const toggle = (id) => setTasks(tasks.map((t) => (t.id === id ? { ...t, done: !t.done } : t)));
  const clearDone = () => setTasks(tasks.filter((t) => !t.done));

  const shown = tasks.filter((t) => (filter === 'pending' ? !t.done : filter === 'done' ? t.done : true));

  return (
    <section>
      <div role="group" aria-label="Filter assignments">
        {['all', 'pending', 'done'].map((f) => (
          <button key={f} type="button" aria-pressed={filter === f} onClick={() => setFilter(f)}>{f}</button>
        ))}
      </div>
      <ul style={{ listStyle: 'none', paddingLeft: 0 }}>
        {shown.map((t) => (
          <li key={t.id}>
            <label>
              <input type="checkbox" checked={t.done} onChange={() => toggle(t.id)} /> {t.title}
            </label>
          </li>
        ))}
      </ul>
      {shown.length === 0 && <p>No assignments here.</p>}
      <button type="button" onClick={clearDone}>Clear completed</button>
    </section>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<AssignmentTracker />);

Live result

{ ...t, done: !t.done } copies the task and changes one field, so the original object is never modified.
