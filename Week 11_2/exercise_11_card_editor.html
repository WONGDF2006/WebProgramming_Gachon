E11 · Live Card Editor

Two controlled inputs feed a Card component through props and children, updating as you type.

function Card({ title, children }) {
  return (
    <section style={{ padding: '16px 20px', border: '1px solid #cbd5e1', borderRadius: 12, background: '#fff' }}>
      <h2 style={{ margin: '0 0 8px', fontSize: '1.15rem' }}>{title || 'Untitled'}</h2>
      {children}
    </section>
  );
}

function CardEditor() {
  const [title, setTitle] = useState('My card');
  const [text, setText] = useState('Edit me above.');

  return (
    <div>
      <p><label htmlFor="card-title">Title </label>
        <input id="card-title" type="text" value={title} onChange={(e) => setTitle(e.target.value)} /></p>
      <p><label htmlFor="card-text">Text </label>
        <input id="card-text" type="text" value={text} onChange={(e) => setText(e.target.value)} /></p>
      <Card title={title}><p style={{ margin: 0 }}>{text}</p></Card>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<CardEditor />);
